        * GET /preferences - Get current user's notification preferences.
        */
       async function handleGetPreferences(
         event: APIGatewayProxyEventV2WithJWTAuthorizer
       ): Promise<APIGatewayProxyResultV2> {
         const userId = event.requestContext.authorizer?.jwt?.claims?.sub as string;

         if (!userId) {
           return badRequest('User ID not found in token');
         }

         const preferences = await getPreferences(docClient, userId);

         // Return without PK/SK (internal fields)
         return ok({
           userId: preferences.userId,
           channels: preferences.channels,
           phone: preferences.phone,
           createdAt: preferences.createdAt,
           updatedAt: preferences.updatedAt,
         });
       }

       /**
        * PUT /preferences - Update current user's notification preferences.
        */
       async function handleUpdatePreferences(
         event: APIGatewayProxyEventV2WithJWTAuthorizer
       ): Promise<APIGatewayProxyResultV2> {
         const userId = event.requestContext.authorizer?.jwt?.claims?.sub as string;

         if (!userId) {
           return badRequest('User ID not found in token');
         }

         // Parse request body
         let body: unknown;
         try {
           body = JSON.parse(event.body ?? '{}');
         } catch {
           return badRequest('Invalid JSON in request body');
         }

         // Validate input
         const validation = PreferencesUpdateSchema.safeParse(body);
         if (!validation.success) {
           return badRequest('Validation failed', validation.error.flatten().fieldErrors);
         }

         const { channels, phone } = validation.data;

         // Validate and format phone number if provided
         let formattedPhone: string | undefined;
         if (phone) {
           const parsed = parsePhoneNumberFromString(phone, 'US');
           if (!parsed || !parsed.isValid()) {
             return badRequest('Invalid phone number. Please use a valid format like +1 555 123 4567');
           }
           formattedPhone = parsed.format('E.164');
         } else if (phone === null) {
           // Explicit null means remove phone
           formattedPhone = undefined;
         }

         // If enabling SMS, require a valid phone
         if (channels?.sms && !formattedPhone) {
           // Check existing phone
           const existing = await getPreferences(docClient, userId);
           if (!existing.phone) {
             return badRequest('Phone number required to enable SMS notifications');
           }
         }

         const updated = await updatePreferences(
           docClient,
           userId,
           channels ?? {},
           formattedPhone
         );

         return ok({
           userId: updated.userId,
           channels: updated.channels,
           phone: updated.phone,
           createdAt: updated.createdAt,
           updatedAt: updated.updatedAt,
         });
       }
       ```

       import { useState, useEffect } from 'react';
       import { parsePhoneNumberFromString, AsYouType } from 'libphonenumber-js';
       import type { NotificationChannels } from '../../types/preferences';

       interface Props {
         channels: NotificationChannels;
         phone?: string;
         onSave: (channels: NotificationChannels, phone?: string | null) => Promise<void>;
         isSaving: boolean;
       }

       export function NotificationPreferences({ channels: initialChannels, phone: initialPhone, onSave, isSaving }: Props) {
         const [channels, setChannels] = useState<NotificationChannels>(initialChannels);
         const [phone, setPhone] = useState(initialPhone ?? '');
         const [phoneError, setPhoneError] = useState('');
         const [isDirty, setIsDirty] = useState(false);

         // Reset state when props change (after save)
         useEffect(() => {
           setChannels(initialChannels);
           setPhone(initialPhone ?? '');
           setIsDirty(false);
         }, [initialChannels, initialPhone]);

         const formatPhoneInput = (value: string): string => {
           const formatter = new AsYouType('US');
           return formatter.input(value);
         };

         const validatePhone = (value: string): boolean => {
           if (!value && !channels.sms) {
             setPhoneError('');
             return true;
           }

           if (!value && channels.sms) {
             setPhoneError('Phone number required for SMS notifications');
             return false;
           }

           const parsed = parsePhoneNumberFromString(value, 'US');
           if (!parsed || !parsed.isValid()) {
             setPhoneError('Please enter a valid phone number');
             return false;
           }

           setPhoneError('');
           return true;
         };

         const handlePhoneChange = (e: React.ChangeEvent<HTMLInputElement>) => {
           const formatted = formatPhoneInput(e.target.value);
           setPhone(formatted);
           setPhoneError('');
           setIsDirty(true);
         };

         const handleChannelToggle = (channel: keyof NotificationChannels) => {
           const newValue = !channels[channel];

           // If enabling SMS without phone, show error
           if (channel === 'sms' && newValue && !phone) {
             setPhoneError('Phone number required for SMS notifications');
           } else if (channel === 'sms' && !newValue) {
             setPhoneError('');
           }

           setChannels(prev => ({ ...prev, [channel]: newValue }));
           setIsDirty(true);
         };

         const handleSave = async () => {
           if (!validatePhone(phone)) return;

           const parsed = phone ? parsePhoneNumberFromString(phone, 'US') : null;
           const formattedPhone = parsed?.format('E.164') ?? (phone ? undefined : null);

           await onSave(channels, formattedPhone);
           setIsDirty(false);
         };

         const channelConfig = [
           {
             key: 'email' as const,
             label: 'Email Notifications',
             description: 'Receive lead alerts via email',
           },
           {
             key: 'slack' as const,
             label: 'Slack Notifications',
             description: 'Receive lead alerts in the team Slack channel',
           },
           {
             key: 'sms' as const,
             label: 'SMS Notifications',
             description: 'Receive lead alerts via text message',
           },
         ];

         return (
           <div className="bg-white rounded-lg shadow p-6">
             <h2 className="text-lg font-semibold text-gray-900 mb-4">Notification Preferences</h2>
             <p className="text-sm text-gray-600 mb-6">
               Choose how you want to be notified when new leads arrive.
             </p>

             <div className="space-y-4">
               {channelConfig.map(({ key, label, description }) => (
                 <div key={key} className="flex items-center justify-between py-3 border-b border-gray-100 last:border-0">
                   <div>
                     <p className="font-medium text-gray-900">{label}</p>
                     <p className="text-sm text-gray-500">{description}</p>
                   </div>
                   <button
                     type="button"
                     role="switch"
                     aria-checked={channels[key]}
                     onClick={() => handleChannelToggle(key)}
                     className={`
                       relative inline-flex h-6 w-11 items-center rounded-full transition-colors
                       ${channels[key] ? 'bg-blue-600' : 'bg-gray-200'}
                     `}
                   >
                     <span
                       className={`
                         inline-block h-4 w-4 transform rounded-full bg-white transition-transform
                         ${channels[key] ? 'translate-x-6' : 'translate-x-1'}
                       `}
                     />
                   </button>
                 </div>
               ))}

               {/* Phone number input - shown when SMS is enabled or phone exists */}
               {(channels.sms || phone) && (
                 <div className="pt-4">
                   <label htmlFor="phone" className="block text-sm font-medium text-gray-700 mb-1">
                     Phone Number for SMS
                   </label>
                   <input
                     type="tel"
                     id="phone"
                     placeholder="+1 (555) 123-4567"
                     value={phone}
                     onChange={handlePhoneChange}
                     onBlur={() => validatePhone(phone)}
                     className={`
                       w-full px-3 py-2 border rounded-lg text-gray-900
                       focus:outline-none focus:ring-2 focus:ring-blue-500
                       ${phoneError ? 'border-red-500' : 'border-gray-300'}
                     `}
                   />
                   {phoneError && (
                     <p className="mt-1 text-sm text-red-600">{phoneError}</p>
                   )}
                   <p className="mt-1 text-xs text-gray-500">
                     US and international numbers supported
                   </p>
                 </div>
               )}
             </div>

             <div className="mt-6 flex items-center justify-between">
               <p className="text-sm text-gray-500">
                 {isDirty ? 'You have unsaved changes' : 'All changes saved'}
               </p>
               <button
                 onClick={handleSave}
                 disabled={isSaving || !isDirty || !!phoneError}
                 className={`
                   px-4 py-2 rounded-lg font-medium transition-colors
                   ${isSaving || !isDirty || phoneError
                     ? 'bg-gray-100 text-gray-400 cursor-not-allowed'
                     : 'bg-blue-600 text-white hover:bg-blue-700'}
                 `}
               >
                 {isSaving ? 'Saving...' : 'Save Changes'}
               </button>
             </div>
           </div>
         );
       }
       ```

    5. Create `admin/src/pages/SettingsPage.tsx`:
       ```typescript
       import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
       import { preferencesApi } from '../api/preferences';
       import { NotificationPreferences } from '../components/settings/NotificationPreferences';
       import type { NotificationChannels, PreferencesUpdateRequest } from '../types/preferences';

       export function SettingsPage() {
         const queryClient = useQueryClient();

         const {
           data: preferences,
           isLoading,
           error,
         } = useQuery({
           queryKey: ['preferences'],
           queryFn: preferencesApi.get,
         });

         const mutation = useMutation({
           mutationFn: (data: PreferencesUpdateRequest) => preferencesApi.update(data),
           onSuccess: () => {
             queryClient.invalidateQueries({ queryKey: ['preferences'] });
           },
         });

         const handleSave = async (channels: NotificationChannels, phone?: string | null) => {
           await mutation.mutateAsync({ channels, phone });
         };

         if (isLoading) {
           return (
             <div className="flex items-center justify-center py-12">
               <div className="text-gray-500">Loading preferences...</div>
             </div>
           );
         }

         if (error) {
           return (
             <div className="bg-red-50 text-red-700 p-4 rounded-lg">
               Failed to load preferences. Please try again.
             </div>
           );
         }

         return (
           <div className="max-w-2xl">
             <h1 className="text-2xl font-bold text-gray-900 mb-6">Settings</h1>

             {preferences && (
               <NotificationPreferences
                 channels={preferences.channels}
                 phone={preferences.phone}
                 onSave={handleSave}
                 isSaving={mutation.isPending}
               />
             )}

             {mutation.isError && (
               <div className="mt-4 bg-red-50 text-red-700 p-4 rounded-lg">
                 Failed to save preferences. Please try again.
               </div>
             )}
           </div>
         );
       }
       ```
